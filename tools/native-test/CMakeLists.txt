cmake_minimum_required(VERSION 3.18)
project(convtest C)
set(LIBREDWG_LIBONLY ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(DISABLE_WERROR ON CACHE BOOL "" FORCE)
set(ENABLE_LTO OFF CACHE BOOL "" FORCE)
set(LIBREDWG_DISABLE_JSON ON CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_Iconv ON)
add_subdirectory(${LIBREDWG_DIR} libredwg)
add_executable(convtest main.c ${CONV_DIR}/convert.c)
target_include_directories(convtest PRIVATE ${CONV_DIR} ${LIBREDWG_DIR}/src ${CMAKE_BINARY_DIR}/libredwg/src)
target_link_libraries(convtest redwg m)
